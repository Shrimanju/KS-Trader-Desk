# Build 1.9.48: dynamic pre-fill entry refresh, AUTO structure TP and confirmed reversal exits.
# Adaptive hybrid trailing is a visual/backtest stop manager for M1 charts.
# Exactly two strategy legs are retained: Entry 1 midpoint and optional Entry 2 20% retracement.
# The old chart add-on Entry 3/4/5 marker logic is removed.
# Entry 2 uses the same master SL/TP and its own allocated lot size.

# Entry 1 is the confirmation candle midpoint (50%).
# Entry 2 is the optional 20% split entry; no Entry 3+ add-ons.
# Reporting is per signal group: only an entry fill activates the group; favorable movement alone remains pending/inactive.
# Inactive is finalized only after BOTH pending entries resolve without either entry triggering.
# This indicator is visual/backtest only; it does not send broker orders.

# ============================================================
# HIGHER TIMEFRAME CONFIRMATION
# ============================================================

@sec_context
def ConfirmHtfEma(self):
    ema20 = Ema.new(self.close, 20)
    ema50 = Ema.new(self.close, 50)
    return ema20[1], ema50[1]


@sec_context
def ConfirmCandle(self):
    average_volume = (
        self.volume[2] + self.volume[3] + self.volume[4]
        + self.volume[5] + self.volume[6] + self.volume[7]
        + self.volume[8] + self.volume[9] + self.volume[10]
        + self.volume[11]
    ) / 10.0

    return (
        self.open[1],
        self.close[1],
        self.volume[1],
        average_volume
    )

    def _auto_rr_for_setup(self, side: str, entry_price: float, risk_distance: float, candle_range: float) -> float:
        # Use the previous ten completed candles only; exclude the current setup candle.
        prior_high = max([
            self.high[1], self.high[2], self.high[3], self.high[4], self.high[5],
            self.high[6], self.high[7], self.high[8], self.high[9], self.high[10]
        ])
        prior_low = min([
            self.low[1], self.low[2], self.low[3], self.low[4], self.low[5],
            self.low[6], self.low[7], self.low[8], self.low[9], self.low[10]
        ])
        target_buffer = max(candle_range * 0.15, risk_distance * 0.03)
        available_rr = 0.0
        if risk_distance <= 0.0:
            return 0.0
        if side == 'buy':
            available_rr = (prior_high - entry_price - target_buffer) / risk_distance
        else:
            available_rr = (entry_price - prior_low - target_buffer) / risk_distance
        if available_rr >= 4.0:
            return 4.0
        elif available_rr >= 3.0:
            return 3.0
        elif available_rr >= 2.0:
            return 2.0
        elif available_rr >= 1.5:
            return 1.5
        elif available_rr >= 1.0:
            return 1.0
        return 0.0

                if candidate_is_valid:
                    if is_buy_trail:
                        if isnan(trail_candidate[0]) or proposed_sl > trail_candidate[0]:
                            trail_candidate[0] = proposed_sl
                            trail_confirm_count[0] = 0.0
                    else:
                        if isnan(trail_candidate[0]) or proposed_sl < trail_candidate[0]:
                            trail_candidate[0] = proposed_sl
                            trail_confirm_count[0] = 0.0

                if not isnan(trail_candidate[0]):
                    candidate_still_safe = (
                        self.close[0] > trail_candidate[0]
                        if is_buy_trail else
                        self.close[0] < trail_candidate[0]
                    )
                    if candidate_still_safe:
                        trail_confirm_count[0] = trail_confirm_count[0] + 1.0
                    else:
                        trail_candidate[0] = nan
                        trail_confirm_count[0] = 0.0

                if (
                    not isnan(trail_candidate[0])
                    and trail_confirm_count[0] >= 3.0
                    and (
                        (is_buy_trail and trail_candidate[0] > sl[0])
                        or ((not is_buy_trail) and trail_candidate[0] < sl[0])
                    )
                ):
                    sl[0] = trail_candidate[0]
                    trail_last_update_bar[0] = float(self.bar_index)
                    trail_confirm_count[0] = 0.0
                    trail_candidate[0] = nan

                    # Draw each label inside its own branch. Indie variables
                    # declared inside an indented branch are not visible outside it.
                    if tp_progress >= 0.95:
                        trail_stage[0] = 3.0
                    elif tp_progress >= 0.85:
                        trail_stage[0] = 2.0
                    elif r_progress >= 1.0:
                        trail_stage[0] = 1.0
                    else:
                        trail_stage[0] = 4.0

            # ================================================
            # EARLY EXIT WARNING
            # ================================================

            if side[0] == 1.0 and early_exit_warned[0] == 0.0:

                buy_failure_score = 0

                if self.close[0] < ema20[0] and ema20[0] < ema50[0]:
                    buy_failure_score += 1

                if htf_sell:
                    buy_failure_score += 1

                if rsi[0] < 45 and rsi[0] < rsi[1]:
                    buy_failure_score += 1

                if bearish_price_action:
                    buy_failure_score += 1

                if buy_failure_score >= 3:

                    early_exit_warned[0] = 1.0

                    if self.is_closed_bar:

                        pass

            elif side[0] == -1.0 and early_exit_warned[0] == 0.0:

                sell_failure_score = 0

                if self.close[0] > ema20[0] and ema20[0] > ema50[0]:
                    sell_failure_score += 1

                if htf_buy:
                    sell_failure_score += 1

                if rsi[0] > 55 and rsi[0] > rsi[1]:
                    sell_failure_score += 1

                if bullish_price_action:
                    sell_failure_score += 1

                if sell_failure_score >= 3:

                    early_exit_warned[0] = 1.0

                    if self.is_closed_bar:

                        pass

            # ================================================
            # ARM NEW BREAKOUT PULLBACK SETUPS
            # ================================================

            if side[0] == 0.0 and not trade_exited:

                if bullish_breakout_volume_setup:

                    pending_buy[0] = 1.0
                    pending_buy_average[0] = (
                        self.high[0] + self.low[0]
                    ) / 2.0
                    pending_buy_low[0] = self.low[0]

                    pending_sell_high[0] = nan

                                report_inactive_buy_mfe_03[0] = report_inactive_buy_mfe_03[0] + 1.0
                            elif setup_unfilled_mfe_bucket[0] == 4.0:

                    calculated_lot = 0.0

                        calculated_lot = int(raw_lot * 100.0) / 100.0

                            calculated_lot * risk_per_one_lot
                        )

                    secondary_price = self.low[0] + candle_range * 0.20
                    midpoint_sl_percent = (sl_distance / entry_price * 100.0) if entry_price > 0.0 else 999.0
                    secondary_sl_percent = (abs(secondary_price - stop_price) / secondary_price * 100.0) if secondary_price > 0.0 else 999.0

                    # completely disable the 20% level. The secondary order

                    midpoint_lot = calculated_lot

                    if calculated_lot >= 0.03:
                        midpoint_lot = int(calculated_lot * 0.60 * 100.0 + 0.5) / 100.0
                        secondary_lot = int((calculated_lot - midpoint_lot) * 100.0 + 0.5) / 100.0
                    elif calculated_lot >= 0.02:
                        midpoint_lot = int(calculated_lot * 0.50 * 100.0 + 0.5) / 100.0

                    if secondary_lot < 0.01 or secondary_sl_percent > secondary_max_sl_percent:

                    if (
                        risk > 0.0
                        and sl_distance <= maximum_allowed_sl_distance
                        and midpoint_sl_percent <= midpoint_max_sl_percent
                        and calculated_lot >= 0.01
                        and rr_multiplier >= 1.0
                    ):

                        entry_rr_multiplier = rr_multiplier
                        if self.risk_reward == 'AUTO':

                                'sell', entry_price, risk, candle_range

                        # Save the planned trade as pending. It becomes active
                        # only when a later completed candle reaches entry.
                        setup_active_counted[0] = 0.0
                        setup_exit_counted[0] = 0.0
                        setup_inactive_counted[0] = 0.0
                        setup_mfe_counted[0] = 0.0
                        setup_unfilled_pnl[0] = 0.0
                        setup_unfilled_leg_count[0] = 0.0
                        setup_unfilled_day[0] = nan
                        setup_unfilled_mfe_bucket[0] = 0.0
                        setup_unfilled_mfe_side[0] = 0.0
                        pending_trade_side[0] = -1.0
                        pending_trade_entry[0] = entry_price
                        pending_trade_sl[0] = stop_price
                        pending_trade_tp[0] = target_price
                        pending_trade_lot[0] = midpoint_lot
                        second_entry_price[0] = secondary_price if secondary_lot >= 0.01 else nan
                        if secondary_lot >= 0.01:
                            pending_second_side[0] = -1.0
                            pending_second_entry[0] = secondary_price
                            pending_second_sl[0] = stop_price
                            pending_second_tp[0] = target_price
                            pending_second_lot[0] = secondary_lot
                            pending_second_day[0] = current_ist_day_key
                            pending_second_bar[0] = float(self.bar_index)
                            pending_second_setup_low[0] = self.low[0]
                            pending_second_setup_high[0] = self.high[0]
                        else:
                            pending_second_side[0] = 0.0
                            pending_second_entry[0] = nan
                            pending_second_sl[0] = nan
                            pending_second_tp[0] = nan
                            pending_second_lot[0] = nan
                            pending_second_day[0] = nan
                            pending_second_bar[0] = -1.0
                            pending_second_setup_low[0] = nan
                            pending_second_setup_high[0] = nan
                        pending_trade_risk_cash[0] = calculated_risk_cash * (midpoint_lot / calculated_lot) if calculated_lot > 0.0 else nan
                        pending_trade_setup_day[0] = current_ist_day_key
                        pending_trade_setup_bar[0] = float(self.bar_index)
                        pending_trade_setup_low[0] = self.low[0]
                        pending_trade_setup_high[0] = self.high[0]
                        pending_trade_max_favorable[0] = nan
                        if current_ist_day_key == self._selected_report_day_key:
                            trade_table_time_10[0] = trade_table_time_9[0]
                            trade_table_lot50_10[0] = trade_table_lot50_9[0]
                            trade_table_lot20_10[0] = trade_table_lot20_9[0]
                            trade_table_sl_10[0] = trade_table_sl_9[0]
                            trade_table_tp_10[0] = trade_table_tp_9[0]
                            trade_table_actual_sl_10[0] = trade_table_actual_sl_9[0]
                            trade_table_actual_tp_10[0] = trade_table_actual_tp_9[0]
                            trade_table_time_9[0] = trade_table_time_8[0]
                            trade_table_lot50_9[0] = trade_table_lot50_8[0]
                            trade_table_lot20_9[0] = trade_table_lot20_8[0]
                            trade_table_sl_9[0] = trade_table_sl_8[0]
                            trade_table_tp_9[0] = trade_table_tp_8[0]
                            trade_table_actual_sl_9[0] = trade_table_actual_sl_8[0]
                            trade_table_actual_tp_9[0] = trade_table_actual_tp_8[0]
                            trade_table_time_8[0] = trade_table_time_7[0]
                            trade_table_lot50_8[0] = trade_table_lot50_7[0]
                            trade_table_lot20_8[0] = trade_table_lot20_7[0]
                            trade_table_sl_8[0] = trade_table_sl_7[0]
                            trade_table_tp_8[0] = trade_table_tp_7[0]
                            trade_table_actual_sl_8[0] = trade_table_actual_sl_7[0]
                            trade_table_actual_tp_8[0] = trade_table_actual_tp_7[0]
                            trade_table_time_7[0] = trade_table_time_6[0]
                            trade_table_lot50_7[0] = trade_table_lot50_6[0]
                            trade_table_lot20_7[0] = trade_table_lot20_6[0]
                            trade_table_sl_7[0] = trade_table_sl_6[0]
                            trade_table_tp_7[0] = trade_table_tp_6[0]
                            trade_table_actual_sl_7[0] = trade_table_actual_sl_6[0]
                            trade_table_actual_tp_7[0] = trade_table_actual_tp_6[0]
                            trade_table_time_6[0] = trade_table_time_5[0]
                            trade_table_lot50_6[0] = trade_table_lot50_5[0]
                            trade_table_lot20_6[0] = trade_table_lot20_5[0]
                            trade_table_sl_6[0] = trade_table_sl_5[0]
                            trade_table_tp_6[0] = trade_table_tp_5[0]
                            trade_table_actual_sl_6[0] = trade_table_actual_sl_5[0]
                            trade_table_actual_tp_6[0] = trade_table_actual_tp_5[0]
                            trade_table_time_5[0] = trade_table_time_4[0]
                            trade_table_lot50_5[0] = trade_table_lot50_4[0]
                            trade_table_lot20_5[0] = trade_table_lot20_4[0]
                            trade_table_sl_5[0] = trade_table_sl_4[0]
                            trade_table_tp_5[0] = trade_table_tp_4[0]
                            trade_table_actual_sl_5[0] = trade_table_actual_sl_4[0]

                            trade_table_time_4[0] = trade_table_time_3[0]

                            trade_table_sl_3[0] = trade_table_sl_2[0]

                            table_lot_sum = midpoint_lot + secondary_lot
                            table_sl_dollars = (
                                abs(entry_price - stop_price) * table_lot_sum
                                * self.contract_size
                                + self.commission_per_lot * table_lot_sum
                            )
                            table_tp_dollars = (
                                abs(target_price - entry_price) * table_lot_sum
                                * self.contract_size
                                - self.commission_per_lot * table_lot_sum
                            )
                            table_time_seconds = int((self.time[0] + 19800.0) % 86400.0)
                            trade_table_time_1[0] = float(table_time_seconds)
                            trade_table_lot50_1[0] = midpoint_lot
                            trade_table_lot20_1[0] = secondary_lot
                            trade_table_sl_1[0] = table_sl_dollars
                            trade_table_tp_1[0] = table_tp_dollars
                            trade_table_actual_sl_1[0] = nan
                            trade_table_actual_tp_1[0] = nan


                        sell_signal = True

                        pending_buy[0] = 0.0
                        pending_buy_average[0] = nan
                        pending_buy_low[0] = nan
                        pending_buy_bar[0] = -1.0

                        pending_sell[0] = 0.0
                        pending_sell_average[0] = nan
                        pending_sell_high[0] = nan
                        pending_sell_bar[0] = -1.0

                        if not (
                            self.delete_setup == 'Primary SELL'
                            and self.bar_index == self.delete_bar_index
                        ):

                            if self.is_closed_bar:

                                pass

                elif buy_entry_ready:

                    recent_low = min([
                        self.low[0], self.low[1],
                        self.low[2], self.low[3],
                        self.low[4], self.low[5],
                        self.low[6], self.low[7]
                    ])

                    # Original Entry 1 logic: confirmation candle midpoint.
                    entry_price = candidate_entry_price

                    if buy_breakout_pullback_ready:
                        buy_reason = 'BREAKOUT + VOLUME PULLBACK'

                    stop_price = recent_low

                    entry_stop_buffer = max(
                        candle_range * 0.25,
                        abs(entry_price) * 0.00005
                    )

                    stop_price = min(
                        stop_price,
                        entry_price - entry_stop_buffer
                    )
